function dfs(r, c) {
if (r<0 || r>=R || c<0 || c>=C || board[r][c] !== 'E') {
return;
}
let mines = 0;
for (const [dr, dc] of dirs) {
if (board[r+dr][c+dc] === 'M') mines++;
}
if (mines > 0) {
board[r][c] = mines.toString();
} else {
board[r][c] = 'B';
for (const [dr, dc] of dirs) {
dfs(r + dr, c + dc);
}
}
}
